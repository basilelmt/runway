# Synchronisation bancaire (Enable Banking) - plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Récupérer automatiquement les opérations La Banque Postale via l'API Enable Banking et les intégrer dans Runway par le pipeline d'import existant, sans validation manuelle.

**Architecture:** Le pipeline d'import (`server/src/services/import.ts`) est découpé en lecture d'une `Source` et traitement `buildPreview`. Un module `server/src/bank/` fournit un client Enable Banking typé derrière une interface `BankProvider` injectable. Un service `bank-sync.ts` orchestre connexion, liaison des comptes et synchronisation, puis appelle `buildPreview` et `commitImport`. Le client React ajoute une carte dans Import, une section dans Paramètres et un déclenchement à l'ouverture.

**Tech Stack:** TypeScript, Hono, Drizzle + better-sqlite3, zod 4, React 19, TanStack Query, vitest. Aucune dépendance ajoutée (JWT signé avec `node:crypto`).

**Spec:** `docs/superpowers/specs/2026-10-04-bank-sync-design.md`

```ts
import { and, asc, eq } from 'drizzle-orm';
import type { Db } from '../db/client';
import { bankAccounts, bankConnections } from '../db/schema';
import { NotFoundError } from '../errors';

export type BankConnection = typeof bankConnections.$inferSelect;
export type BankAccount = typeof bankAccounts.$inferSelect;

export function listConnections(db: Db): BankConnection[] {
  return db.select().from(bankConnections).orderBy(asc(bankConnections.id)).all();
}

export function getConnection(db: Db, id: number): BankConnection {
  const row = db.select().from(bankConnections).where(eq(bankConnections.id, id)).get();
  if (!row) throw new NotFoundError('Connexion bancaire introuvable');
  return row;
}

export function createConnection(db: Db, input: { aspspName: string; aspspCountry: string; state: string }): BankConnection {
  return db.insert(bankConnections).values(input).returning().get();
}

export function updateConnection(db: Db, id: number, patch: Partial<Omit<BankConnection, 'id' | 'createdAt'>>): BankConnection {
  const row = db.update(bankConnections).set(patch).where(eq(bankConnections.id, id)).returning().get();
  if (!row) throw new NotFoundError('Connexion bancaire introuvable');
  return row;
}

/** Une seule autorisation à la fois : les connexions jamais terminées sont jetées. */
export function deletePendingConnections(db: Db): void {
  db.delete(bankConnections).where(eq(bankConnections.status, 'pending')).run();
}

export function listBankAccounts(db: Db, connectionId: number): BankAccount[] {
  return db.select().from(bankAccounts).where(eq(bankAccounts.connectionId, connectionId)).orderBy(asc(bankAccounts.id)).all();
}

export function getBankAccount(db: Db, id: number): BankAccount {
  const row = db.select().from(bankAccounts).where(eq(bankAccounts.id, id)).get();
  if (!row) throw new NotFoundError('Compte bancaire introuvable');
  return row;
}

/**
 * Enregistre un compte renvoyé par la banque. Un compte déjà connu de la connexion (même IBAN,
 * sinon même uid) garde sa liaison et ses dates : seuls `uid` et `bankName` sont rafraîchis.
 */
export function upsertBankAccount(
  db: Db,
  input: { connectionId: number; uid: string; iban: string | null; bankName: string; accountId: number | null },
): BankAccount {
  const byIban = input.iban
    ? db.select().from(bankAccounts).where(and(eq(bankAccounts.connectionId, input.connectionId), eq(bankAccounts.iban, input.iban))).get()
    : undefined;
  const existing = byIban ?? db.select().from(bankAccounts).where(and(eq(bankAccounts.connectionId, input.connectionId), eq(bankAccounts.uid, input.uid))).get();
  if (existing) {
    return db
      .update(bankAccounts)
      .set({ uid: input.uid, bankName: input.bankName, iban: input.iban ?? existing.iban, accountId: existing.accountId ?? input.accountId })
      .where(eq(bankAccounts.id, existing.id))
      .returning()
      .get();
  }
  return db.insert(bankAccounts).values(input).returning().get();
}

export function updateBankAccount(db: Db, id: number, patch: Partial<Omit<BankAccount, 'id' | 'connectionId'>>): BankAccount {
  const row = db.update(bankAccounts).set(patch).where(eq(bankAccounts.id, id)).returning().get();
  if (!row) throw new NotFoundError('Compte bancaire introuvable');
  return row;
}
```

```ts
/** Date de la dernière opération confirmée touchant un compte (débitée ou créditée), ou null. */
export function lastConfirmedDate(db: Db, accountId: number): string | null {
  const row = db
    .select({ date: transactions.date })
    .from(transactions)
    .where(and(eq(transactions.status, 'confirmed'), or(eq(transactions.accountId, accountId), eq(transactions.toAccountId, accountId))))
    .orderBy(desc(transactions.date))
    .limit(1)
    .get();
  return row?.date ?? null;
}

/**
 * Une opération confirmée existe-t-elle sur ce compte, ce jour, pour ce montant signé ?
 * Sert à la jonction entre l'historique importé des relevés et la synchronisation API,
 * dont les libellés diffèrent (l'empreinte ne suffit pas).
 */
export function hasConfirmedOn(db: Db, accountId: number, date: string, signedAmount: number): boolean {
  const side =
    signedAmount < 0
      ? and(eq(transactions.accountId, accountId), inArray(transactions.kind, ['expense', 'transfer']))
      : or(and(eq(transactions.accountId, accountId), eq(transactions.kind, 'income')), and(eq(transactions.toAccountId, accountId), eq(transactions.kind, 'transfer')));
  const row = db
    .select({ id: transactions.id })
    .from(transactions)
    .where(and(eq(transactions.status, 'confirmed'), eq(transactions.date, date), eq(transactions.amount, Math.abs(signedAmount)), side))
    .limit(1)
    .get();
  return row !== undefined;
}
```

`and`, `or`, `desc`, `eq`, `inArray` sont déjà importés en tête du fichier.

```ts
import { createSign } from 'node:crypto';

const b64url = (s: string) => Buffer.from(s).toString('base64url');

/** JWT RS256 attendu par Enable Banking : `kid` = Application ID, valable une heure. */
export function signJwt(appId: string, privateKeyPem: string, nowSeconds = Math.floor(Date.now() / 1000)): string {
  const header = b64url(JSON.stringify({ typ: 'JWT', alg: 'RS256', kid: appId }));
  const payload = b64url(JSON.stringify({ iss: 'enablebanking.com', aud: 'api.enablebanking.com', iat: nowSeconds, exp: nowSeconds + 3600 }));
  const signature = createSign('RSA-SHA256').update(`${header}.${payload}`).sign(privateKeyPem, 'base64url');
  return `${header}.${payload}.${signature}`;
}
```

```ts
import { z } from 'zod';
import { signJwt } from './jwt';
import { bankTransactionSchema, BankProviderError, type Aspsp, type BankProvider, type BankTransaction, type SessionInfo } from './provider';

export type PsuInfo = { ip: string; userAgent: string };

export type EnableBankingOptions = {
  appId: string;
  privateKey: string;
  /** Utilisateur présent : ces en-têtes évitent de consommer le quota DSP2 des accès sans présence. */
  psu: PsuInfo;
  fetch?: typeof fetch;
  baseUrl?: string;
};

const DEFAULT_BASE_URL = 'https://api.enablebanking.com';

const aspspsSchema = z.object({ aspsps: z.array(z.object({ name: z.string(), country: z.string(), maximum_consent_validity: z.number().nullish() })) });
const authSchema = z.object({ url: z.string() });
const sessionSchema = z.object({
  session_id: z.string(),
  access: z.object({ valid_until: z.string().nullish() }).nullish(),
  accounts: z.array(
    z.object({
      uid: z.string(),
      account_id: z.object({ iban: z.string().nullish() }).nullish(),
      name: z.string().nullish(),
      product: z.string().nullish(),
    }),
  ),
});
const sessionStatusSchema = z.object({ status: z.string(), access: z.object({ valid_until: z.string().nullish() }).nullish() });
const transactionsSchema = z.object({ transactions: z.array(bankTransactionSchema), continuation_key: z.string().nullish() });
const applicationSchema = z.object({ name: z.string(), environment: z.string() });

export class EnableBankingClient implements BankProvider {
  private readonly fetchFn: typeof fetch;
  private readonly baseUrl: string;

  constructor(private readonly opts: EnableBankingOptions) {
    this.fetchFn = opts.fetch ?? globalThis.fetch;
    this.baseUrl = opts.baseUrl ?? DEFAULT_BASE_URL;
  }

  private async request<T>(method: string, path: string, schema: z.ZodType<T>, body?: unknown): Promise<T> {
    const res = await this.fetchFn(`${this.baseUrl}${path}`, {
      method,
      headers: {
        authorization: `Bearer ${signJwt(this.opts.appId, this.opts.privateKey)}`,
        accept: 'application/json',
        'content-type': 'application/json',
        'psu-ip-address': this.opts.psu.ip,
        'psu-user-agent': this.opts.psu.userAgent,
      },
      body: body === undefined ? undefined : JSON.stringify(body),
    });
    const text = await res.text();
    let data: unknown = null;
    try {
      data = text ? JSON.parse(text) : {};
    } catch {
      throw new BankProviderError(`Réponse illisible d'Enable Banking (${res.status})`, res.status);
    }
    if (!res.ok) {
      const err = (data ?? {}) as { message?: unknown; error?: unknown; code?: unknown };
      const message = typeof err.message === 'string' ? err.message : typeof err.error === 'string' ? err.error : `Enable Banking a répondu ${res.status}`;
      throw new BankProviderError(message, res.status, typeof err.code === 'string' ? err.code : null);
    }
    const parsed = schema.safeParse(data);
    if (!parsed.success) throw new BankProviderError(`Réponse inattendue d'Enable Banking sur ${path}`, res.status);
    return parsed.data;
  }

  async getApplication(): Promise<{ name: string; environment: string }> {
    return this.request('GET', '/application', applicationSchema);
  }

  async getMaxConsentSeconds(aspsp: Aspsp): Promise<number | null> {
    const { aspsps } = await this.request('GET', `/aspsps?country=${encodeURIComponent(aspsp.country)}`, aspspsSchema);
    return aspsps.find((a) => a.name === aspsp.name)?.maximum_consent_validity ?? null;
  }

  async startAuth(input: { aspsp: Aspsp; validUntil: string; redirectUrl: string; state: string }): Promise<{ url: string }> {
    const { url } = await this.request('POST', '/auth', authSchema, {
      access: { valid_until: input.validUntil },
      aspsp: input.aspsp,
      state: input.state,
      redirect_url: input.redirectUrl,
      psu_type: 'personal',
    });
    return { url };
  }

  async createSession(code: string): Promise<SessionInfo> {
    const s = await this.request('POST', '/sessions', sessionSchema, { code });
    return {
      sessionId: s.session_id,
      validUntil: s.access?.valid_until ?? null,
      accounts: s.accounts.map((a) => ({ uid: a.uid, iban: a.account_id?.iban ?? null, name: a.name?.trim() || a.product?.trim() || 'Compte' })),
    };
  }

  async getSession(sessionId: string): Promise<{ status: string; validUntil: string | null }> {
    const s = await this.request('GET', `/sessions/${encodeURIComponent(sessionId)}`, sessionStatusSchema);
    return { status: s.status, validUntil: s.access?.valid_until ?? null };
  }

  async listTransactions(uid: string, dateFrom: string, dateTo: string): Promise<BankTransaction[]> {
    const all: BankTransaction[] = [];
    let continuation: string | null | undefined;
    do {
      const params = new URLSearchParams({ date_from: dateFrom, date_to: dateTo, transaction_status: 'BOOK' });
      if (continuation) params.set('continuation_key', continuation);
      const page = await this.request('GET', `/accounts/${encodeURIComponent(uid)}/transactions?${params}`, transactionsSchema);
      for (const tx of page.transactions) if (!tx.status || tx.status === 'BOOK') all.push(tx);
      continuation = page.continuation_key;
    } while (continuation);
    return all;
  }

  async deleteSession(sessionId: string): Promise<void> {
    await this.request('DELETE', `/sessions/${encodeURIComponent(sessionId)}`, z.unknown());
  }
}
```

```ts
function rowText(row: ImportPreviewRow): string {
  return `${row.label} ${row.note ?? ''}`;
}

/**
 * Virements entre comptes propres : un débit « VIREMENT » avec un crédit du même montant à ±4 jours
 * chez un compte frère devient un transfert vers ce compte. Quand le libellé cite un IBAN, il désigne
 * la cible ; sans IBAN (cas de l'API), un unique compte frère portant la contrepartie suffit.
 */
function detectInternalTransfers(siblings: Sibling[], rows: ImportPreviewRow[]): void {
  const known = siblings.filter((s) => s.accountId !== null);
  if (known.length === 0) return;
  for (const row of rows) {
    const text = rowText(row);
    if (row.status !== 'new' || row.signedAmount >= 0 || !/VIREMENT/i.test(text)) continue;
    const compact = text.replace(/\s/g, '').toUpperCase();
    const hasCounterpart = (s: Sibling) => s.rows.some((r) => r.signedAmount === -row.signedAmount && Math.abs(daysBetween(row.date, r.date)) <= TRANSFER_TOLERANCE_DAYS);
    const byIban = known.find((s) => s.iban && compact.includes(s.iban));
    const candidates = byIban ? [byIban] : IBAN_RE.test(compact) ? [] : known.filter(hasCounterpart);
    const target = candidates.length === 1 ? candidates[0]! : null;
    if (!target || !hasCounterpart(target)) continue;
    row.status = 'transfer';
    row.transferToAccountId = target.accountId;
    row.envelopeId = null;
  }
}

/** Crédit d'un virement dont le transfert est déjà saisi (importé depuis l'autre compte) : à ignorer. */
function markInternalCredits(db: Db, accountId: number, rows: ImportPreviewRow[]): void {
  const claimed = new Set<number>();
  for (const row of rows) {
    if (row.status !== 'new' || row.signedAmount <= 0 || !/VIREMENT/i.test(rowText(row))) continue;
    const transfers = findTransfersTo(db, accountId, row.signedAmount, addDays(row.date, -TRANSFER_TOLERANCE_DAYS), addDays(row.date, TRANSFER_TOLERANCE_DAYS));
    const free = transfers.find((t) => !claimed.has(t.id));
    if (!free) continue;
    claimed.add(free.id);
    row.status = 'internal';
  }
}

/**
 * Traitement commun à toutes les sources : empreintes, doublons, virements internes,
 * rapprochement des opérations prévues et règles de tri.
 */
export function buildPreview(db: Db, source: Source, accountId: number | null): ImportPreview {
  if (accountId != null) getAccount(db, accountId);

  const rules = listSortingRules(db);
  const planned = accountId == null ? [] : listPlanned(db, accountId);

  // Deux lignes identiques le même jour (deux cafés) doivent garder deux empreintes distinctes.
  const seen = new Map<string, number>();
  const rows: ImportPreviewRow[] = source.rows.map((r, index) => {
    const key = `${r.date}|${r.signedAmount}|${r.label}`;
    const occurrence = seen.get(key) ?? 0;
    seen.set(key, occurrence + 1);
    return {
      index,
      date: r.date,
      label: r.label,
      signedAmount: r.signedAmount,
      status: 'new',
      fingerprint: fingerprint(accountId ?? 0, r.date, r.signedAmount, r.label, occurrence),
      plannedId: null,
      plannedLabel: null,
      envelopeId: r.signedAmount < 0 ? applySortingRules(r.label, rules) : null,
      note: r.note,
      transferToAccountId: null,
    };
  });

  const existing = accountId == null ? new Set<string>() : findByFingerprints(db, rows.map((r) => r.fingerprint));
  for (const row of rows) if (existing.has(row.fingerprint)) row.status = 'duplicate';

  if (accountId != null) {
    detectInternalTransfers(source.siblings, rows);
    markInternalCredits(db, accountId, rows);
  }

  const fresh = rows.filter((r) => r.status === 'new');
  const matches = matchPlanned(fresh, planned);
  const plannedById = new Map(planned.map((p) => [p.id, p]));
  for (const row of fresh) {
    const plannedId = matches.get(row.index);
    if (plannedId === undefined) continue;
    const p = plannedById.get(plannedId)!;
    row.status = 'matched';
    row.plannedId = plannedId;
    row.plannedLabel = p.label;
    row.envelopeId = p.envelopeId;
  }

  const count = (status: ImportPreviewRow['status']) => rows.filter((r) => r.status === status).length;
  return {
    parserId: source.parserId,
    bankAccountNumber: source.bankAccountNumber,
    accountId,
    rows,
    counts: { total: rows.length, new: count('new'), duplicate: count('duplicate'), matched: count('matched'), transfer: count('transfer'), internal: count('internal') },
    sections: source.sections,
    sectionIndex: source.sectionIndex,
  };
}

  if (accountId == null && source.bankAccountNumber) {

  }
  return buildPreview(db, source, accountId);
}
```

```ts
import type { SourceRow } from '../services/import';
import type { BankTransaction } from './provider';

/** « 12.34 » → 1234, « -12,5 » → -1250, sans passer par un flottant. */
export function parseAmountToCents(amount: string): number {
  const m = /^(-?)(\d+)(?:[.,](\d{1,2}))?$/.exec(amount.trim());
  if (!m) throw new Error(`Montant illisible : ${amount}`);
  const cents = Number(m[2]) * 100 + Number((m[3] ?? '').padEnd(2, '0'));
  return m[1] === '-' ? -cents : cents;
}

export type MappedRow = { ok: true; row: SourceRow } | { ok: false; reason: string };

const ISO_DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

/**
 * Opération Enable Banking → ligne de source. Libellé : référence de paiement, sinon contrepartie,
 * sinon description du code bancaire. La note garde les détails annexes pour la recherche et les virements.
 */
export function toSourceRow(tx: BankTransaction): MappedRow {
  const date = [tx.booking_date, tx.value_date, tx.transaction_date].find((d): d is string => !!d && ISO_DATE_RE.test(d)) ?? null;
  if (!date) return { ok: false, reason: 'date absente' };

  let cents: number;
  try {
    cents = parseAmountToCents(tx.transaction_amount.amount);
  } catch {
    return { ok: false, reason: 'montant illisible' };
  }
  if (cents === 0) return { ok: false, reason: 'montant nul' };
  const signedAmount = tx.credit_debit_indicator === 'DBIT' ? -Math.abs(cents) : Math.abs(cents);

  const counterparty = (tx.credit_debit_indicator === 'DBIT' ? tx.creditor?.name : tx.debtor?.name)?.trim() || null;
  const remittance = (tx.remittance_information ?? []).map((s) => s.trim()).filter(Boolean).join(' ');
  const label = (remittance || counterparty || tx.bank_transaction_code?.description?.trim() || 'Opération').slice(0, 200);

  const extras = [
    counterparty && counterparty !== label ? counterparty : null,
    tx.entry_reference ? `Réf. ${tx.entry_reference}` : null,
    tx.value_date && tx.value_date !== date ? `Valeur ${tx.value_date}` : null,
  ].filter((s): s is string => s !== null);
  const note = extras.length > 0 ? `${label} · ${extras.join(' · ')}` : null;

  return { ok: true, row: { date, label, signedAmount, note } };
}
```

```ts
import { randomBytes } from 'node:crypto';
import type { BankAccountView, BankAuthStart, BankConnectionView, BankSyncConnectionResult, BankSyncRunResult, BankSyncState } from '@shared/api-types';
import { addDays, compareISODate, type ISODate } from '@shared/dates';
import type { BankLinkInput } from '@shared/schemas';
import { toSourceRow } from '../bank/mapping';
import type { Aspsp, BankProvider } from '../bank/provider';
import type { Db } from '../db/client';
import { BadRequestError } from '../errors';
import { findAccountByBankNumber, getAccount } from '../repos/accounts';
import {
  createConnection,
  deletePendingConnections,
  getBankAccount,
  getConnection,
  listBankAccounts,
  listConnections,
  updateBankAccount,
  updateConnection,
  upsertBankAccount,
  type BankAccount,
  type BankConnection,
} from '../repos/bank';
import { hasConfirmedOn, lastConfirmedDate } from '../repos/transactions';
import { buildPreview, commitImport, type Sibling, type SourceRow } from './import';

export const ASPSP: Aspsp = { name: 'La Banque Postale', country: 'FR' };
/** Déclarée telle quelle dans l'application Enable Banking. Le navigateur ne la charge pas : l'utilisateur colle l'URL. */
export const REDIRECT_URL = 'https://localhost:3000/bank-sync/callback';

const SYNC_FILE_NAME = 'Synchronisation La Banque Postale';
const PARSER_ID = 'enablebanking';
const DEFAULT_CONSENT_SECONDS = 90 * 86_400;
/** Les banques datent parfois en retard : on relit les derniers jours, l'empreinte absorbe le chevauchement. */
const OVERLAP_DAYS = 7;
/** Après la date de départ, fenêtre où une opération déjà connue (date + montant) est un doublon de jonction. */
const JUNCTION_DAYS = 10;
const EMPTY_ACCOUNT_HISTORY_DAYS = 90;

function toAccountView(a: BankAccount): BankAccountView {
  return { id: a.id, uid: a.uid, iban: a.iban, bankName: a.bankName, accountId: a.accountId, syncFrom: a.syncFrom, lastSyncedDate: a.lastSyncedDate };
}

function toConnectionView(db: Db, c: BankConnection): BankConnectionView {
  return {
    id: c.id,
    aspspName: c.aspspName,
    status: c.status,
    validUntil: c.validUntil,
    lastSyncAt: c.lastSyncAt,
    lastError: c.lastError,
    accounts: listBankAccounts(db, c.id).map(toAccountView),
  };
}

export function getState(db: Db, configured: boolean): BankSyncState {
  return {
    configured,
    redirectUrl: REDIRECT_URL,
    connections: listConnections(db)
      .filter((c) => c.status !== 'pending')
      .map((c) => toConnectionView(db, c)),
  };
}

/** Numéro de compte français (11 caractères) contenu dans un IBAN FR, pour retrouver un compte Runway. */
function frenchAccountNumber(iban: string | null): string | null {
  const compact = iban?.replace(/\s/g, '').toUpperCase() ?? '';
  return compact.startsWith('FR') && compact.length === 27 ? compact.slice(14, 25) : null;
}

export async function startConnection(db: Db, provider: BankProvider, input: { connectionId?: number; now: Date }): Promise<BankAuthStart> {
  const state = randomBytes(16).toString('hex');
  const maxSeconds = (await provider.getMaxConsentSeconds(ASPSP)) ?? DEFAULT_CONSENT_SECONDS;
  const validUntil = new Date(input.now.getTime() + maxSeconds * 1000).toISOString();
  const { url } = await provider.startAuth({ aspsp: ASPSP, validUntil, redirectUrl: REDIRECT_URL, state });

  let connection: BankConnection;
  if (input.connectionId !== undefined) {
    // Renouvellement : la connexion garde son statut et ses comptes, seul le state change.
    connection = updateConnection(db, getConnection(db, input.connectionId).id, { state });
  } else {
    deletePendingConnections(db);
    connection = createConnection(db, { aspspName: ASPSP.name, aspspCountry: ASPSP.country, state });
  }
  return { connectionId: connection.id, url };
}

export async function completeConnection(db: Db, provider: BankProvider, input: { connectionId: number; code: string; state: string }): Promise<BankConnectionView> {
  const connection = getConnection(db, input.connectionId);
  if (connection.state !== input.state) throw new BadRequestError("Retour d'autorisation invalide : relance la connexion");

  const session = await provider.createSession(input.code);
  const known = listBankAccounts(db, connection.id);
  for (const account of session.accounts) {
    const suggested = frenchAccountNumber(account.iban);
    const accountId = suggested ? (findAccountByBankNumber(db, suggested)?.id ?? null) : null;
    upsertBankAccount(db, { connectionId: connection.id, uid: account.uid, iban: account.iban, bankName: account.name, accountId });
  }
  // Première connexion : les liaisons restent à confirmer. Renouvellement : la synchro peut reprendre.
  const status = known.some((a) => a.syncFrom !== null) ? 'active' : 'authorized';
  const updated = updateConnection(db, connection.id, { sessionId: session.sessionId, validUntil: session.validUntil, status, lastError: null });
  return toConnectionView(db, updated);
}

export function linkAccounts(db: Db, input: { connectionId: number; links: BankLinkInput['links']; today: ISODate }): BankConnectionView {
  const connection = getConnection(db, input.connectionId);
  db.transaction((tx) => {
    const t = tx as unknown as Db;
    for (const link of input.links) {
      const bank = getBankAccount(t, link.bankAccountId);
      if (bank.connectionId !== connection.id) throw new BadRequestError("Ce compte n'appartient pas à cette connexion");
      if (link.accountId === null) {
        updateBankAccount(t, bank.id, { accountId: null, syncFrom: null });
        continue;
      }
      getAccount(t, link.accountId);
      const last = lastConfirmedDate(t, link.accountId);
      const syncFrom = last ? addDays(last, 1) : addDays(input.today, -EMPTY_ACCOUNT_HISTORY_DAYS);
      updateBankAccount(t, bank.id, { accountId: link.accountId, syncFrom, lastSyncedDate: null });
    }
    if (connection.status === 'authorized') updateConnection(t, connection.id, { status: 'active' });
  });
  return toConnectionView(db, getConnection(db, connection.id));
}

export async function disconnect(db: Db, provider: BankProvider, connectionId: number): Promise<void> {
  const connection = getConnection(db, connectionId);
  if (connection.sessionId) {
    try {
      await provider.deleteSession(connection.sessionId);
    } catch {
      // Session déjà expirée ou révoquée côté banque : l'état local suffit.
    }
  }
  updateConnection(db, connection.id, { status: 'revoked', lastError: null });
}

function applyJunctionDedup(db: Db, bank: BankAccount, rows: ReturnType<typeof buildPreview>['rows']): void {
  const limit = addDays(bank.syncFrom!, JUNCTION_DAYS);
  for (const row of rows) {
    if (row.status !== 'new' || compareISODate(row.date, limit) > 0) continue;
    if (hasConfirmedOn(db, bank.accountId!, row.date, row.signedAmount)) row.status = 'duplicate';
  }
}

async function syncConnection(db: Db, provider: BankProvider, connection: BankConnection, today: ISODate, now: Date): Promise<BankSyncConnectionResult> {
  const result: BankSyncConnectionResult = { connectionId: connection.id, status: connection.status, created: 0, matched: 0, duplicates: 0, skipped: 0, error: null };
  try {
    const session = await provider.getSession(connection.sessionId!);
    const expired = session.status !== 'AUTHORIZED' || (session.validUntil !== null && new Date(session.validUntil).getTime() < now.getTime());
    if (expired) {
      updateConnection(db, connection.id, { status: 'expired', lastError: null });
      result.status = 'expired';
      return result;
    }

    const linked = listBankAccounts(db, connection.id).filter((a) => a.accountId !== null && a.syncFrom !== null && !getAccount(db, a.accountId).archived);
    const fetched: { bank: BankAccount; rows: SourceRow[] }[] = [];
    for (const bank of linked) {
      const overlapFrom = bank.lastSyncedDate ? addDays(bank.lastSyncedDate, -OVERLAP_DAYS) : bank.syncFrom!;
      const from = compareISODate(overlapFrom, bank.syncFrom!) > 0 ? overlapFrom : bank.syncFrom!;
      const rows: SourceRow[] = [];
      for (const tx of await provider.listTransactions(bank.uid, from, today)) {
        const mapped = toSourceRow(tx);
        if (mapped.ok) rows.push(mapped.row);
        else result.skipped++;
      }
      fetched.push({ bank, rows });
    }

    for (const { bank, rows } of fetched) {
      const siblings: Sibling[] = fetched.filter((f) => f.bank.id !== bank.id).map((f) => ({ accountId: f.bank.accountId, iban: f.bank.iban, rows: f.rows }));
      const preview = buildPreview(db, { parserId: PARSER_ID, bankAccountNumber: null, rows, sections: [], sectionIndex: 0, siblings }, bank.accountId);
      applyJunctionDedup(db, bank, preview.rows);
      const toWrite = preview.rows.filter((r) => r.status === 'new' || r.status === 'matched' || r.status === 'transfer');
      const duplicates = preview.rows.length - toWrite.length;
      if (toWrite.length > 0) {
        const record = commitImport(db, {
          fileName: SYNC_FILE_NAME,
          accountId: bank.accountId!,
          duplicateCount: duplicates,
          rows: toWrite.map(({ index, date, label, signedAmount, fingerprint, plannedId, envelopeId, note, transferToAccountId }) => ({ index, date, label, signedAmount, fingerprint, plannedId, envelopeId, note, transferToAccountId })),
        });
        result.created += record.newCount;
        result.matched += record.matchedCount;
      }
      result.duplicates += duplicates;
      const latest = rows.reduce<string | null>((max, r) => (max === null || compareISODate(r.date, max) > 0 ? r.date : max), bank.lastSyncedDate);
      if (latest !== bank.lastSyncedDate) updateBankAccount(db, bank.id, { lastSyncedDate: latest });
    }
    updateConnection(db, connection.id, { lastSyncAt: now.toISOString(), lastError: null });
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    updateConnection(db, connection.id, { lastSyncAt: now.toISOString(), lastError: message });
    result.error = message;
  }
  return result;
}

let running: Promise<BankSyncRunResult> | null = null;

/**
 * Synchronise toutes les connexions actives. Une seule exécution à la fois : un second appel
 * pendant qu'une synchro tourne reçoit le même résultat.
 */
export function runSync(db: Db, provider: BankProvider, input: { today: ISODate; now: Date; onlyIfStaleMinutes?: number }): Promise<BankSyncRunResult> {
  if (running) return running;
  running = (async () => {
    const active = listConnections(db).filter((c) => c.status === 'active' && c.sessionId !== null);
    if (active.length === 0) return { reason: 'no_connection' as const, connections: [] };
    if (input.onlyIfStaleMinutes !== undefined) {
      const latest = Math.max(...active.map((c) => (c.lastSyncAt ? new Date(c.lastSyncAt).getTime() : 0)));
      if (input.now.getTime() - latest < input.onlyIfStaleMinutes * 60_000) return { reason: 'fresh' as const, connections: [] };
    }
    const connections: BankSyncConnectionResult[] = [];
    for (const c of active) connections.push(await syncConnection(db, provider, c, input.today, input.now));
    return { reason: 'ok' as const, connections };
  })().finally(() => {
    running = null;
  });
  return running;
}
```

```ts
import { Hono, type Context } from 'hono';
import { z } from 'zod';
import { bankCompleteSchema, bankLinkSchema, bankRunSchema, bankStartSchema, type Settings } from '@shared/schemas';
import type { AppEnv } from '../app';
import { EnableBankingClient, type PsuInfo } from '../bank/enable-banking';
import { BankProviderError, type BankProvider } from '../bank/provider';
import { AppError, BadRequestError } from '../errors';
import { getSettings } from '../repos/settings';
import { completeConnection, disconnect, getState, linkAccounts, runSync, startConnection } from '../services/bank-sync';
import { validate } from '../validate';

export type BankDeps = {
  /** Fabrique du fournisseur, injectable pour les tests. */
  provider?: (settings: Settings, psu: PsuInfo) => BankProvider;
  now?: () => Date;
};

const idParam = z.object({ id: z.coerce.number().int().positive() });

const defaultProvider: NonNullable<BankDeps['provider']> = (s, psu) => new EnableBankingClient({ appId: s.enableBankingAppId, privateKey: s.enableBankingPrivateKey, psu });

function isConfigured(s: Settings): boolean {
  return s.enableBankingAppId !== '' && s.enableBankingPrivateKey !== '';
}

function psuInfo(c: Context<AppEnv>): PsuInfo {
  const forwarded = c.req.header('x-forwarded-for')?.split(',')[0]?.trim();
  return { ip: forwarded || '127.0.0.1', userAgent: c.req.header('user-agent') ?? 'Runway' };
}

/** Une erreur du fournisseur devient une 502 lisible plutôt qu'une 500 anonyme. */
async function guard<T>(fn: () => Promise<T>): Promise<T> {
  try {
    return await fn();
  } catch (err) {
    if (err instanceof BankProviderError) throw new AppError(err.message, 502, 'bank_provider');
    if (err instanceof AppError) throw err;
    if (err instanceof Error) throw new AppError(err.message, 502, 'bank_provider');
    throw err;
  }
}

export const bankSyncRoutes = (deps: BankDeps = {}) => {
  const makeProvider = deps.provider ?? defaultProvider;
  const now = deps.now ?? (() => new Date());

  function provider(c: Context<AppEnv>): BankProvider {
    const settings = getSettings(c.get('db'));
    if (!isConfigured(settings)) throw new BadRequestError('Identifiants Enable Banking manquants : renseigne-les dans Paramètres, section Banque');
    return makeProvider(settings, psuInfo(c));
  }

  return new Hono<AppEnv>()
    .get('/', (c) => c.json(getState(c.get('db'), isConfigured(getSettings(c.get('db'))))))
    .post('/test', async (c) => c.json(await guard(() => provider(c).getApplication())))
    .post('/connections', validate('json', bankStartSchema), async (c) => {
      const { connectionId } = c.req.valid('json');
      const p = provider(c);
      return c.json(await guard(() => startConnection(c.get('db'), p, { connectionId: connectionId ?? undefined, now: now() })), 201);
    })
    .post('/connections/:id/complete', validate('param', idParam), validate('json', bankCompleteSchema), async (c) => {
      const p = provider(c);
      const { code, state } = c.req.valid('json');
      return c.json(await guard(() => completeConnection(c.get('db'), p, { connectionId: c.req.valid('param').id, code, state })));
    })
    .put('/connections/:id/accounts', validate('param', idParam), validate('json', bankLinkSchema), (c) => {
      return c.json(linkAccounts(c.get('db'), { connectionId: c.req.valid('param').id, links: c.req.valid('json').links, today: c.get('today') }));
    })
    .post('/run', validate('json', bankRunSchema), async (c) => {
      const settings = getSettings(c.get('db'));
      if (!isConfigured(settings)) return c.json({ reason: 'not_configured', connections: [] });
      const p = makeProvider(settings, psuInfo(c));
      return c.json(await runSync(c.get('db'), p, { today: c.get('today'), now: now(), onlyIfStaleMinutes: c.req.valid('json').onlyIfStaleMinutes }));
    })
```

```tsx
/** Extrait `code` et `state` de l'URL de retour collée (ou d'une simple query string). */
function parseReturn(text: string): { code: string; state: string } | null {
  const trimmed = text.trim();
  if (!trimmed) return null;
  let params: URLSearchParams;
  try {
    params = new URL(trimmed).searchParams;
  } catch {
    params = new URLSearchParams(trimmed.replace(/^\?/, ''));
  }
  const code = params.get('code');
  const state = params.get('state');
  return code && state ? { code, state } : null;
}

  onDisconnect,
}: {
  connection: BankConnectionView;
  accounts: Account[];
  syncing: boolean;
  onSync: () => void;
  onRenew: () => void;
  renewing: boolean;
  onDisconnect: () => void;
}) {
  const expired = connection.status === 'expired';
  const expiresIn = connection.validUntil ? daysUntil(connection.validUntil) : null;
  const expiringSoon = !expired && expiresIn !== null && expiresIn <= EXPIRY_WARNING_DAYS;
  const linked = connection.accounts.filter((a) => a.accountId !== null);
  const accountName = (id: number) => accounts.find((a) => a.id === id)?.name ?? `Compte ${id}`;

  return (
    <>
      <div className={styles.meta}>
        <span>{connection.lastSyncAt ? `Dernière synchronisation ${formatDateTime(connection.lastSyncAt)}` : 'Jamais synchronisé'}</span>
        {connection.validUntil ? <span className="muted">· consentement jusqu'au {formatDateLong(connection.validUntil.slice(0, 10))}</span> : null}
      </div>
      <div className={styles.accounts}>
        {linked.map((a) => (
          <div key={a.id} className={styles.accountRow}>
            <span className={styles.bankName}>
              <strong>{a.bankName}</strong>
              {a.iban ? <span className={['mono', styles.iban].join(' ')}>{a.iban}</span> : null}
            </span>
            <ArrowRight className={styles.arrow} aria-hidden />
            <span>
              {accountName(a.accountId!)}
              {a.lastSyncedDate ? <span className="muted"> · à jour au {formatDateLong(a.lastSyncedDate)}</span> : null}
            </span>
          </div>
        ))}
      </div>
      {connection.lastError ? (
        <p className={styles.error} role="alert">
          Dernière synchronisation en échec : {connection.lastError}
        </p>
      ) : null}
      {expired ? (
        <p className={styles.error}>Le consentement a expiré : la banque n'envoie plus rien tant que tu ne l'as pas renouvelé.</p>
      ) : expiringSoon ? (
        <p className={styles.text}>Le consentement expire {expiresIn! <= 0 ? "aujourd'hui" : `dans ${plural(expiresIn!, 'jour')}`}. Renouvelle-le pour ne pas interrompre la synchronisation.</p>
      ) : null}
      <div className={styles.actions}>
        {expired || expiringSoon ? (
          <Button variant="primary" icon={<RefreshCw aria-hidden />} loading={renewing} onClick={onRenew}>
            Renouveler le consentement
          </Button>
        ) : null}
        {!expired ? (
          <Button variant={expiringSoon ? 'secondary' : 'primary'} icon={<RefreshCw aria-hidden />} loading={syncing} onClick={onSync}>
            Synchroniser maintenant
          </Button>
        ) : null}
        <Button variant="ghost" onClick={onDisconnect}>
          Déconnecter
        </Button>
      </div>
    </>
  );
}
```

Créer `client/src/components/import/useAutoBankSync.ts` :

```ts
import { useEffect, useRef } from 'react';
import { useRunBankSync } from '@/api/hooks';
import { useToast } from '@/components/ui';
import { plural } from '@/lib/format';

const STALE_MINUTES = 60;

/**
 * À l'ouverture de l'app, demande une synchronisation si la dernière date de plus d'une heure.
 * Silencieux sauf quand des opérations arrivent ; les erreurs restent visibles dans la carte Import.
 */
export function useAutoBankSync(): void {
  const toast = useToast();
  const { mutate } = useRunBankSync({
    onSuccess: (r) => {
      if (r.reason !== 'ok') return;
      const created = r.connections.reduce((n, c) => n + c.created, 0);
      const matched = r.connections.reduce((n, c) => n + c.matched, 0);
      if (created + matched === 0) return;
      toast.success(`${plural(created, 'opération synchronisée', 'opérations synchronisées')}${matched ? `, ${matched} rapprochée${matched > 1 ? 's' : ''}` : ''}`);
    },
  });
  const started = useRef(false);
  useEffect(() => {
    if (started.current) return;
    started.current = true;
    mutate(STALE_MINUTES);
  }, [mutate]);
}
```
