# Demander à Claude - plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** un panneau « Demander » où l'on pose une question en français ; Claude interroge la base en lecture seule via un outil MCP, répond avec un chiffre, un graphique et des actions applicables d'un clic.

**Architecture:** côté serveur, `POST /api/ask` lance `claude -p` isolé (prompt système Runway, seul outil `mcp__runway__query`), lit sa sortie `stream-json`, relaie les étapes en SSE, puis valide la réponse structurée : exécution du SQL du graphique, validation et enrichissement des actions.
Le serveur MCP est un script stdio lancé par `claude`, qui ouvre la base en lecture seule.
Côté client, un `AskProvider` garde le fil et l'état d'ouverture ; un panneau non modal affiche le fil, les graphiques et les cartes d'action, appliquées via les hooks existants.

**Tech Stack:** TypeScript, Hono (`streamSSE`), better-sqlite3, `@modelcontextprotocol/sdk` 1.32, zod 4, React 19, TanStack Query, Recharts, Vitest, Testing Library.

Spec : `docs/superpowers/specs/2026-10-08-demander-design.md`.

```ts
import Database from 'better-sqlite3';

/** Connexion en lecture seule : toute écriture échoue au niveau de SQLite, quelle que soit la requête. */
export function openReadOnly(path: string): Database.Database {
  const db = new Database(path, { readonly: true, fileMustExist: true });
  db.pragma('query_only = ON');
  return db;
}

export type QueryResult = { rows: Record<string, unknown>[]; truncated: boolean };

/**
 * Exécute une seule instruction de lecture et renvoie au plus `limit` lignes.
 * `prepare` refuse plusieurs instructions ; `reader` écarte tout ce qui ne renvoie pas de lignes.
 */
export function runReadOnlyQuery(db: Database.Database, sql: string, limit = 200): QueryResult {
  const stmt = db.prepare(sql);
  if (!stmt.reader) throw new Error('Seules les requêtes de lecture (SELECT, WITH) sont permises');
  if (/^\s*pragma/i.test(sql)) throw new Error('PRAGMA non permis');
  const rows: Record<string, unknown>[] = [];
  for (const row of stmt.iterate() as Iterable<Record<string, unknown>>) {
    if (rows.length === limit) return { rows, truncated: true };
    rows.push(row);
  }
  return { rows, truncated: false };
}
```

Note : `better-sqlite3` lève « The supplied SQL string contains more than one statement » pour `select 1; delete ...`. `stmt.iterate()` arrêté tôt libère le curseur à la sortie de la boucle (`return` dans un `for...of`).

```ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import type Database from 'better-sqlite3';
import { z } from 'zod';
import { runReadOnlyQuery } from './query';

/** Serveur MCP de Runway : un seul outil, une requête SQL en lecture seule. */
export function createRunwayMcpServer(db: Database.Database): McpServer {
  const server = new McpServer({ name: 'runway', version: '1.0.0' });
  server.registerTool(
    'query',
    {
      description: 'Exécute une requête SQL de lecture (SELECT ou WITH) sur la base SQLite de Runway. Renvoie au plus 200 lignes en JSON.',
      inputSchema: { sql: z.string().describe('Une seule instruction SELECT ou WITH') },
    },
    async ({ sql }) => {
      try {
        return { content: [{ type: 'text', text: JSON.stringify(runReadOnlyQuery(db, sql)) }] };
      } catch (err) {
        return { content: [{ type: 'text', text: `Erreur SQL : ${(err as Error).message}` }], isError: true };
      }
    },
  );
  return server;
}
```

```ts
import { inArray } from 'drizzle-orm';
import type { AskActionView } from '@shared/api-types';
import { askActionSchema, type AskAction } from '@shared/schemas';
import type { Db } from '../db/client';
import { transactions } from '../db/schema';
import { listAccounts } from '../repos/accounts';
import { listEnvelopes } from '../repos/envelopes';

/** Valide les actions proposées par Claude et les enrichit pour l'affichage ; les invalides sont écartées. */
export function validateActions(db: Db, raw: unknown[]): { actions: AskActionView[]; rejected: number } {
  const envelopes = new Map(listEnvelopes(db).map((e) => [e.id, e]));
  const accounts = new Map(listAccounts(db).map((a) => [a.id, a]));
  const actions: AskActionView[] = [];
  for (const item of raw) {
    const parsed = askActionSchema.safeParse(item);
    const view = parsed.success ? enrich(db, parsed.data, envelopes, accounts) : null;
    if (view) actions.push(view);
  }
  return { actions, rejected: raw.length - actions.length };
}

function enrich(
  db: Db,
  action: AskAction,
  envelopes: Map<number, { id: number; name: string; monthlyTarget: number | null }>,
  accounts: Map<number, { id: number; name: string }>,
): AskActionView | null {
  switch (action.type) {
    case 'sort': {
      const env = envelopes.get(action.envelopeId);
      const ids = [...new Set(action.transactionIds)];
      const rows = db.select({ id: transactions.id, kind: transactions.kind, amount: transactions.amount }).from(transactions).where(inArray(transactions.id, ids)).all();
      if (!env || rows.length !== ids.length || rows.some((r) => r.kind !== 'expense')) return null;
      return { type: 'sort', transactionIds: ids, envelopeId: env.id, envelopeName: env.name, rulePattern: action.rulePattern, count: ids.length, total: rows.reduce((s, r) => s + r.amount, 0) };
    }
    case 'move': {
      const to = envelopes.get(action.toEnvelopeId);
      const from = action.fromEnvelopeId === null ? null : envelopes.get(action.fromEnvelopeId);
      if (!to || from === undefined || from?.id === to.id) return null;
      return { type: 'move', fromEnvelopeId: from?.id ?? null, fromName: from?.name ?? 'À répartir', toEnvelopeId: to.id, toName: to.name, amount: action.amount };
    }
    case 'budget': {
      const env = envelopes.get(action.envelopeId);
      if (!env) return null;
      return { type: 'budget', envelopeId: env.id, envelopeName: env.name, monthlyTarget: action.monthlyTarget, current: env.monthlyTarget };
    }
    case 'recurring': {
      const account = action.rule.accountId == null ? undefined : accounts.get(action.rule.accountId);
      if (!account) return null;
      if (action.rule.envelopeId != null && !envelopes.has(action.rule.envelopeId)) return null;
      return { type: 'recurring', rule: action.rule, accountName: account.name };
    }
  }
}
```

```ts
/** Lance `claude` et transmet sa sortie ligne à ligne (`--output-format stream-json`). 124 = délai dépassé, 130 = interrompu. */
export const spawnClaudeStream: StreamExec = (args, { cwd, timeoutMs, signal, onLine }) =>
  new Promise((resolve) => {
    const child = spawn('claude', args, { cwd, stdio: ['ignore', 'pipe', 'pipe'], env: { ...process.env, CLAUDECODE: '' } });
    let buffer = '';
    let stderr = '';
    let reason: 'timeout' | 'abort' | null = null;
    const kill = (why: 'timeout' | 'abort') => {
      reason = why;
      child.kill('SIGKILL');
    };
    const timer = setTimeout(() => kill('timeout'), timeoutMs);
    const onAbort = () => kill('abort');
    signal.addEventListener('abort', onAbort, { once: true });
    child.stdout.on('data', (d: Buffer) => {
      buffer += d.toString();
      let nl: number;
      while ((nl = buffer.indexOf('\n')) >= 0) {
        const lineText = buffer.slice(0, nl).trim();
        buffer = buffer.slice(nl + 1);
        if (lineText) onLine(lineText);
      }
    });
    child.stderr.on('data', (d: Buffer) => (stderr += d.toString()));
    const done = (exitCode: number) => {
      clearTimeout(timer);
      signal.removeEventListener('abort', onAbort);
      if (buffer.trim()) onLine(buffer.trim());
      resolve({ stderr, exitCode: reason === 'timeout' ? 124 : reason === 'abort' ? 130 : exitCode });
    };
    child.on('error', (err: NodeJS.ErrnoException) => {
      stderr = err.message;
      done(err.code === 'ENOENT' ? 127 : 1);
    });
    child.on('close', (code) => done(code ?? 1));
  });
```

```ts
export function stepText(sql: string): string {
  const s = sql.toLowerCase();
  if (/strftime|group by\s+.*(month|mois|substr)/.test(s)) return 'Je calcule mois par mois…';
  if (s.includes('recurring_rules')) return 'Je regarde tes charges fixes…';
  if (s.includes('envelopes') && !s.includes('transactions')) return 'Je regarde tes enveloppes…';
  if (s.includes('accounts') && !s.includes('transactions')) return 'Je regarde tes comptes…';

  return 'Je vérifie dans tes données…';
```

```ts
export function askArgs(input: { question: string; sessionId: string | null; newSessionId: string; systemPrompt: string; dbPath: string; model?: string }): string[] {
  const mcp = { mcpServers: { runway: { command: tsxBin, args: [mcpScript], env: { RUNWAY_DB: input.dbPath } } } };

  else args.push('--session-id', input.newSessionId);
  if (input.model) args.push('--model', input.model);
  args.push(input.question);

/** Pose la question à Claude et émet les étapes puis la réponse (ou une erreur). Ne lève jamais. */
export async function runAsk(db: Db, input: AskRequest & { today: ISODate }, deps: AskDeps, emit: (e: AskStreamEvent) => void): Promise<void> {
  const newSessionId = (deps.newSessionId ?? randomUUID)();
  const cwd = join(deps.dbPath ? dirname(deps.dbPath) : 'tmp', 'claude-ask');
  mkdirSync(cwd, { recursive: true });
  const args = askArgs({
    question: input.question,
    sessionId: input.sessionId,
    newSessionId,
    systemPrompt: buildAskSystemPrompt({ today: input.today, envelopes: listEnvelopes(db), accounts: listAccounts(db) }),
    dbPath: deps.dbPath,
    model: deps.model,
  });

  let final: { is_error?: boolean; result?: string; structured_output?: unknown; session_id?: string } | null = null;
  try {
    const { exitCode, stderr } = await (deps.stream ?? spawnClaudeStream)(args, {
      cwd,
      timeoutMs: ASK_TIMEOUT_MS,
      signal: deps.signal,
      onLine: (text) => {
        let event: any;
        try {
          event = JSON.parse(text);
        } catch {
          return;
        }
        if (event.type === 'assistant') {
          for (const block of event.message?.content ?? []) {
            if (block.type === 'tool_use' && block.name === 'mcp__runway__query') emit({ event: 'step', data: { text: stepText(String(block.input?.sql ?? '')) } });
          }
        }
        if (event.type === 'result') final = event;
      },
    });
    if (exitCode === 130) return;
    if (exitCode === 127) throw new ClaudeError("Claude Code n'est pas installé (binaire `claude` introuvable)", 'claude_not_installed', 503);
    if (exitCode === 124) throw new ClaudeError('Claude a mis trop de temps à répondre', 'claude_timeout', 504);
    const resultError = final ? claudeErrorFromResult(final) : null;
    if (resultError) throw resultError;
    const parsed = askAnswerSchema.safeParse(final?.structured_output);
    if (!parsed.success) throw new ClaudeError(`Réponse de Claude illisible${stderr ? ` : ${stderr.slice(0, 200)}` : ''}`, 'claude_invalid_output');

    const { actions, rejected } = validateActions(db, parsed.data.actions);
    if (rejected > 0) console.warn(`Demander : ${rejected} action(s) proposée(s) écartée(s)`);
    emit({
      event: 'answer',
      data: {
        sessionId: final?.session_id ?? input.sessionId ?? newSessionId,
        answer: parsed.data.answer,
        highlight: parsed.data.highlight,
        chart: parsed.data.chart ? runChart(db, deps.dbPath, parsed.data.chart) : null,
        actions,
      },
    });
  } catch (err) {
    const e = err instanceof ClaudeError ? err : new ClaudeError((err as Error).message, 'claude_error');
    emit({ event: 'error', data: { message: e.message, code: e.code ?? 'claude_error' } });
  }
}

/** Exécute la requête du graphique en lecture seule ; un format inattendu omet le graphique. */
function runChart(db: Db, dbPath: string, chart: { kind: 'bar' | 'line'; title: string; sql: string }): AskChart | null {
  const conn = dbPath ? openReadOnly(dbPath) : db.$client;
  try {
    const { rows } = runReadOnlyQuery(conn, chart.sql, 60);
    const points = rows.map((r) => ({ label: r.label, value: r.value }));
    if (points.length === 0 || points.some((p) => typeof p.label !== 'string' || typeof p.value !== 'number')) return null;
    return { kind: chart.kind, title: chart.title, rows: points as { label: string; value: number }[] };
  } catch {
    return null;
  } finally {
    if (dbPath) conn.close();
  }
}
```

`client/src/api/ask.ts` :

```ts
import type { AskStreamEvent } from '@shared/api-types';
import type { AskRequest } from '@shared/schemas';
import { ApiError } from './client';

/** Envoie une question et transmet les événements SSE au fil de l'eau. */
export async function streamAsk(req: AskRequest, onEvent: (e: AskStreamEvent) => void, signal: AbortSignal): Promise<void> {
  const res = await fetch('/api/ask', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(req), signal });
  if (!res.ok || !res.body) {
    const body = (await res.json().catch(() => ({}))) as { error?: string; code?: string };
    throw new ApiError(body.error ?? `Erreur ${res.status}`, res.status, body.code);
  }
  const reader = res.body.pipeThrough(new TextDecoderStream()).getReader();
  let buffer = '';
  for (;;) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += value;
    let sep: number;
    while ((sep = buffer.indexOf('\n\n')) >= 0) {
      const block = buffer.slice(0, sep);
      buffer = buffer.slice(sep + 2);
      const event = /^event: (.*)$/m.exec(block)?.[1];
      const data = /^data: (.*)$/m.exec(block)?.[1];
      if (event && data) onEvent({ event, data: JSON.parse(data) } as AskStreamEvent);
    }
  }
}
```
